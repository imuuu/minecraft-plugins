generalstore_script:

    type: assignment



    interact scripts:

    - generalstore_open



    actions:

        on assignment:

        - trigger name:damage state:true

        - trigger name:click state:true

generalstore_open:
    debug: false
    type: interact



    steps:

        'Opening the store':

            click trigger:

                script:

                - execute as_server 'gsopenfor <player.name> %shop%'

            damage trigger:

                script:

                - execute as_server 'gsopenfor <player.name> %shop%'
